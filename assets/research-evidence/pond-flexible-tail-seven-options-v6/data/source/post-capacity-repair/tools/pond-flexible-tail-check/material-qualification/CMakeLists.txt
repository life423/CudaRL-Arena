cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120)
endif()
project(PondFlexibleTailMaterialQualification LANGUAGES CXX CUDA)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(source_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../src")
set(legacy_header "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/flexible-tail/phase-aware-v4/source/src/physics/flexible_tail.h")
if(NOT EXISTS "${legacy_header}")
    message(FATAL_ERROR "The preserved phase-aware-v4 body source is required for exact default comparison")
endif()
configure_file("${legacy_header}" legacy_flexible_tail.h COPYONLY)
add_executable(pond_flexible_tail_material_check check.cu)
set_target_properties(pond_flexible_tail_material_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES CUDA_RUNTIME_LIBRARY Static)
target_compile_features(pond_flexible_tail_material_check PRIVATE cxx_std_17)
target_include_directories(pond_flexible_tail_material_check PRIVATE "${source_root}" "${source_root}/nn" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(pond_flexible_tail_material_check PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
