cmake_minimum_required(VERSION 3.24)
project(PondShorelineWarning LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
set(source_root "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
set(habitat_path "${CMAKE_CURRENT_SOURCE_DIR}/../../flexible-tail-preview/habitat.json")
file(READ "${habitat_path}" habitat)
foreach(dimension IN ITEMS water_radius_x_m water_radius_z_m flat_radius_x_m flat_radius_z_m floor_y_m reference_scale_m)
    string(JSON ${dimension} GET "${habitat}" ${dimension})
endforeach()
configure_file("../pond-flexible-tail-check/habitat_config.h.in" habitat_config.h @ONLY)
add_executable(pond_shoreline_warning audit.cpp "${source_root}/core/checkpoint.cpp")
target_compile_features(pond_shoreline_warning PRIVATE cxx_std_17)
target_compile_definitions(pond_shoreline_warning PRIVATE CUDARL_RELIABLE_CHOOSER=1 CUDARL_APPENDED_COVERAGE=1)
target_include_directories(pond_shoreline_warning PRIVATE "${source_root}" "${source_root}/nn" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(pond_shoreline_warning PRIVATE Threads::Threads)
