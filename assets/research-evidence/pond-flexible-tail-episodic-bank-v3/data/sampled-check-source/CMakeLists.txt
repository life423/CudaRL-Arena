cmake_minimum_required(VERSION 3.24)
project(CriticScan LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(critic_scan scan.cpp ../source/src/core/checkpoint.cpp)
target_compile_features(critic_scan PRIVATE cxx_std_17)
target_compile_definitions(critic_scan PRIVATE CUDARL_RELIABLE_CHOOSER=1 CUDARL_APPENDED_COVERAGE=1 CUDARL_CURVED_BANK=1)
target_include_directories(critic_scan PRIVATE ../source/src ../source/src/nn ../native-build)
add_executable(sampled_check sampled-check.cpp ../source/src/core/checkpoint.cpp)
target_compile_features(sampled_check PRIVATE cxx_std_17)
target_compile_definitions(sampled_check PRIVATE CUDARL_RELIABLE_CHOOSER=1 CUDARL_APPENDED_COVERAGE=1 CUDARL_CURVED_BANK=1)
target_include_directories(sampled_check PRIVATE ../source/src ../source/src/nn ../native-build)
