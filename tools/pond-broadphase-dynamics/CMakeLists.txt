cmake_minimum_required(VERSION 3.24)
project(PondSweptClearance LANGUAGES CXX CUDA)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(exact "${root}/build/clearance-dynamics-v1/native")
set(source_root "${root}")
set(version swept)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
find_package(CUDAToolkit REQUIRED)
file(SHA256 "${exact}/candidate.dir/Release/candidate.lib" accepted_sha)
if(NOT accepted_sha STREQUAL "abd5854a3afc3be8ea8982164287d6d8d84d51f16167193330f8b9b95897e788")
    message(FATAL_ERROR "This comparison requires the exact qualified solver library")
endif()
file(SHA256 "${exact}/baseline.dir/Release/baseline.lib" legacy_sha)
if(NOT legacy_sha STREQUAL "a27da9dc3ffb5a7f3a78744fbf7087496067dbdc8897d51e44184c785ae3af96")
    message(FATAL_ERROR "Unrecognized historical reference")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../pond-clearance-dynamics/capture.cmake")
insert_after("struct StepInfo {" "    double observed_swept_bound=0; bool observed_far=false;")
insert_after("struct SubstepTrace {" "    double observed_swept_bound=0; bool observed_far=false;")
insert_after("trace->contact_iterations=info->contact_iterations;" "        trace->observed_swept_bound=info->observed_swept_bound; trace->observed_far=info->observed_far;")
insert_after("++clearance->rejected_contacts;" "        info.observed_far=true;info.observed_swept_bound=separation.lower_m;")
file(WRITE "${CMAKE_BINARY_DIR}/capture/swept/simulation/pond_runtime.h" "${replay_runtime}")
file(SHA256 "${root}/src/simulation/pond_runtime.h" new_sha)
file(SHA256 "${root}/src/environment/pond_swept_clearance.h" bounds_sha)
file(SHA256 "${root}/src/environment/pond_body_geometry.h" geometry_sha)
if(NOT geometry_sha STREQUAL "19f6ed1777e0464af1b4e583a44837896c2061a11efc63851bf319b01ed5905a")
    message(FATAL_ERROR "Tested exact geometry changed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/src/simulation/pond_runtime.h" "${root}/src/environment/pond_swept_clearance.h" "${root}/src/environment/pond_body_geometry.h")
file(WRITE "${CMAKE_BINARY_DIR}/bindings.h" "#pragma once\nconstexpr char old_sha[]=\"${accepted_sha}\",new_sha[]=\"${new_sha}\",common_sha[]=\"${geometry_sha}\",swept_sha[]=\"${bounds_sha}\";\n")
add_library(swept OBJECT variant.cu)
target_compile_definitions(swept PRIVATE MODEL_NAMESPACE=swept_cudarl)
target_include_directories(swept PRIVATE "${CMAKE_BINARY_DIR}/capture/swept" "${root}/src" "${root}/src/nn")
target_compile_options(swept PRIVATE --fmad=false)
add_executable(pond_broadphase_dynamics broad.cpp "${root}/src/core/checkpoint.cpp" $<TARGET_OBJECTS:swept>)
target_include_directories(pond_broadphase_dynamics PRIVATE "${root}/src" "${CMAKE_BINARY_DIR}")
target_link_libraries(pond_broadphase_dynamics PRIVATE "${exact}/candidate.dir/Release/candidate.lib" "${exact}/baseline.dir/Release/baseline.lib" CUDA::cudart_static bcrypt)
