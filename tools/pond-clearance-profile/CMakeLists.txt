cmake_minimum_required(VERSION 3.24)
project(PondClearanceProfile LANGUAGES CXX CUDA)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(frozen "${root}/build/turn38-residual/source-snapshot")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
find_package(CUDAToolkit REQUIRED)
foreach(version IN ITEMS baseline candidate)
    if(version STREQUAL "baseline")
        set(source_root "${frozen}")
    else()
        set(source_root "${root}")
    endif()
    file(SHA256 "${source_root}/src/environment/pond_body_geometry.h" ${version}_sha)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_root}/src/environment/pond_body_geometry.h")
    add_library(${version} OBJECT variant.cu)
    target_compile_definitions(${version} PRIVATE PROFILE_NAMESPACE=${version} MODEL_NAMESPACE=${version}_cudarl)
    if(version STREQUAL "baseline")
        target_compile_definitions(${version} PRIVATE PROFILE_BASELINE=1)
    endif()
    target_include_directories(${version} PRIVATE "${source_root}/src" "${source_root}/src/nn")
    target_compile_options(${version} PRIVATE --fmad=false)
endforeach()
if(NOT baseline_sha STREQUAL "c4ca21d8c9a080ba44072ea3e2015711601cc23f8f726f520711583391ed0767")
    message(FATAL_ERROR "Invalid frozen geometry source")
endif()
set(contract "")
foreach(relative IN ITEMS cuda/pond_math_profile.h environment/pond_geometry_v1.h simulation/pond_runtime.h
        physics/dynamics.h physics/contact.h nn/motor/movement_model.h nn/motor/joint_motor.h nn/motor/state.h core/hd.h simulation/world.h)
    file(SHA256 "${frozen}/src/${relative}" old)
    file(SHA256 "${root}/src/${relative}" new)
    if(NOT old STREQUAL new)
        message(FATAL_ERROR "Unmatched input dependency: ${relative}")
    endif()
    string(APPEND contract "${relative}:${old};")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${frozen}/src/${relative}" "${root}/src/${relative}")
endforeach()
string(SHA256 contract_sha "${contract}")
file(WRITE "${CMAKE_BINARY_DIR}/bindings.h" "#pragma once\nconstexpr char old_sha[]=\"${baseline_sha}\",new_sha[]=\"${candidate_sha}\",common_sha[]=\"${contract_sha}\";\n")
add_executable(pond_clearance_profile profile.cpp "${root}/src/core/checkpoint.cpp" $<TARGET_OBJECTS:baseline> $<TARGET_OBJECTS:candidate>)
target_include_directories(pond_clearance_profile PRIVATE "${root}/src" "${CMAKE_BINARY_DIR}")
target_link_libraries(pond_clearance_profile PRIVATE CUDA::cudart_static bcrypt)
