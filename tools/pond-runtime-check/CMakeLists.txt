cmake_minimum_required(VERSION 3.24)
project(PondRuntimeCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(pond_runtime_check check.cpp ../../src/core/checkpoint.cpp ../../src/nn/motor/saved_motor_contract.cpp)
target_include_directories(pond_runtime_check PRIVATE ../../src ../../src/nn)
if(MSVC)
  target_compile_options(pond_runtime_check PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(pond_runtime_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
