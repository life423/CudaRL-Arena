cmake_minimum_required(VERSION 3.24)
project(PondObservationCheck LANGUAGES CXX)
get_filename_component(worktree "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(pond_observation_check observation_check.cpp)
add_executable(pond_turn_input_audit turn_input_audit.cpp "${worktree}/src/core/checkpoint.cpp")
target_include_directories(pond_turn_input_audit PRIVATE "${worktree}/src" "${worktree}/src/nn")
target_include_directories(pond_observation_check PRIVATE "${worktree}/src" "${worktree}/src/nn")
if(MSVC)
  target_compile_options(pond_observation_check PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(pond_observation_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
