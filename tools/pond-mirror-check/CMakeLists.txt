cmake_minimum_required(VERSION 3.24)
project(PondMirrorCheck LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
find_package(CUDAToolkit REQUIRED)
add_executable(pond_mirror_check check.cpp replay.cu ../../src/core/checkpoint.cpp)
target_include_directories(pond_mirror_check PRIVATE ../../src ../../src/nn)
target_compile_options(pond_mirror_check PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
target_link_libraries(pond_mirror_check PRIVATE CUDA::cudart_static)
