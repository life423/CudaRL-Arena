cmake_minimum_required(VERSION 3.24)
project(PondPureComponentBench LANGUAGES CXX CUDA)
get_filename_component(worktree "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(FROZEN_ROOT "${worktree}/build/turn32/source-snapshot-v2" CACHE PATH "Frozen v2 sources")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
add_executable(pond_component_bench component.cu "${FROZEN_ROOT}/src/core/checkpoint.cpp")
target_include_directories(pond_component_bench PRIVATE "${FROZEN_ROOT}/src" "${FROZEN_ROOT}/src/nn")
target_compile_options(pond_component_bench PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
set_source_files_properties(component.cu PROPERTIES COMPILE_DEFINITIONS "NOMINMAX;WIN32_LEAN_AND_MEAN")
