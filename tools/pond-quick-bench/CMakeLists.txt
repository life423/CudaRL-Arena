cmake_minimum_required(VERSION 3.24)
project(PondQuickBench LANGUAGES CXX)
get_filename_component(worktree "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(FROZEN_ROOT "${worktree}/build/turn32/source-snapshot-v2" CACHE PATH "Exact frozen v2 source snapshot")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(CUDAToolkit REQUIRED)
set(manifest "")
foreach(path IN ITEMS "${CMAKE_CURRENT_LIST_DIR}/bench.cpp" "${CMAKE_CURRENT_LIST_DIR}/CMakeLists.txt"
    "${FROZEN_ROOT}/src/simulation/pond_runtime.h" "${FROZEN_ROOT}/src/rl/pond_ppo_task.h"
    "${FROZEN_ROOT}/src/rl/pond_ppo_contract.h" "${FROZEN_ROOT}/src/core/checkpoint.h")
    file(SHA256 "${path}" hash)
    string(APPEND manifest "${path} ${hash}\n")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/source-manifest.txt" "${manifest}")
string(SHA256 identity "${manifest}")
file(WRITE "${CMAKE_BINARY_DIR}/quick_identity.h" "#pragma once\ninline constexpr char quick_source_sha256[]=\"${identity}\";\n")
add_executable(pond_quick_bench bench.cpp "${FROZEN_ROOT}/src/core/checkpoint.cpp")
target_include_directories(pond_quick_bench PRIVATE "${FROZEN_ROOT}/src" "${FROZEN_ROOT}/src/nn" "${CMAKE_BINARY_DIR}")
target_compile_options(pond_quick_bench PRIVATE /W4 /WX /permissive-)
set_source_files_properties(bench.cpp PROPERTIES COMPILE_DEFINITIONS "NOMINMAX;WIN32_LEAN_AND_MEAN")
target_link_libraries(pond_quick_bench PRIVATE CUDA::cuda_driver bcrypt)
