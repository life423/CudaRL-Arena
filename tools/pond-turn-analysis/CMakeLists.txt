cmake_minimum_required(VERSION 3.24)
project(PondRecordedTurnAnalysis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(pond_turn_analysis analyze.cpp)
target_compile_definitions(pond_turn_analysis PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(pond_turn_analysis PRIVATE /W4 /WX /permissive-)
target_link_libraries(pond_turn_analysis PRIVATE bcrypt)
