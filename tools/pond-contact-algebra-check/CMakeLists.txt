cmake_minimum_required(VERSION 3.24)
project(PondContactAlgebraCheck LANGUAGES CXX)
get_filename_component(worktree "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(pond_contact_algebra_check contact_algebra_check.cpp)
target_include_directories(pond_contact_algebra_check PRIVATE "${worktree}/src" "${worktree}/src/nn")
if(MSVC)
  target_compile_options(pond_contact_algebra_check PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(pond_contact_algebra_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
