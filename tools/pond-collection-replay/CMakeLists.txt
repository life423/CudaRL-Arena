cmake_minimum_required(VERSION 3.24)
project(PondCollectionReplay LANGUAGES CXX CUDA)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
find_package(CUDAToolkit REQUIRED)
add_subdirectory(../pond-residual-ppo "${CMAKE_BINARY_DIR}/residual" EXCLUDE_FROM_ALL)
add_executable(pond_collection_replay driver.cpp replay.cu ../pond-ppo/admission.cpp "${root}/src/core/checkpoint.cpp" "${root}/src/nn/motor/saved_motor_contract.cpp")
target_include_directories(pond_collection_replay PRIVATE "${CMAKE_BINARY_DIR}")
target_compile_definitions(pond_collection_replay PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:NOMINMAX> $<$<COMPILE_LANGUAGE:CUDA>:WIN32_LEAN_AND_MEAN>)
set_property(SOURCE replay.cu APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_BINARY_DIR}/residual_identity.h;${CMAKE_CURRENT_LIST_DIR}/../pond-residual-ppo/train.cu;${CMAKE_CURRENT_LIST_DIR}/../pond-ppo/train.cu")
target_compile_options(pond_collection_replay PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
target_link_libraries(pond_collection_replay PRIVATE pond_ppo_gpu_core bcrypt)
add_custom_target(qualified_native_bins DEPENDS pond_collection_replay pond_residual_ppo)
