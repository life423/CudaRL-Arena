cmake_minimum_required(VERSION 3.24)
project(PondGapGradientCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(pond_gap_gradient_check gap_gradient_check.cpp ../../src/core/checkpoint.cpp)
target_include_directories(pond_gap_gradient_check PRIVATE ../../src ../../src/nn)
if(MSVC)
  target_compile_options(pond_gap_gradient_check PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(pond_gap_gradient_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
