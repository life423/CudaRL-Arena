cmake_minimum_required(VERSION 3.24)
project(MovingStartUpstream LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(READ "${repo}/flexible-tail-preview/habitat.json" habitat)
foreach(field IN ITEMS water_radius_x_m water_radius_z_m flat_radius_x_m flat_radius_z_m floor_y_m reference_scale_m)
    string(JSON ${field} GET "${habitat}" ${field})
endforeach()
configure_file("${repo}/tools/pond-flexible-tail-check/habitat_config.h.in" habitat_config.h @ONLY)
add_executable(upstream upstream.cpp "${repo}/src/core/checkpoint.cpp")
target_compile_features(upstream PRIVATE cxx_std_17)
target_include_directories(upstream PRIVATE "${repo}/src" "${repo}/src/nn" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(upstream PRIVATE CUDARL_RELIABLE_CHOOSER=1 CUDARL_APPENDED_COVERAGE=1 CUDARL_THREE_DEPTHS=1)
