cmake_minimum_required(VERSION 3.24)
project(PondChooserQualification LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
set(chooser "${CMAKE_CURRENT_SOURCE_DIR}/../pond-flexible-tail-chooser")
set(native "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
set(habitat_path "${CMAKE_CURRENT_SOURCE_DIR}/../../flexible-tail-preview/habitat.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${habitat_path}")
file(READ "${habitat_path}" habitat)
string(JSON schema GET "${habitat}" schema)
if(NOT schema EQUAL 1)
    message(FATAL_ERROR "Unsupported habitat descriptor")
endif()
foreach(field IN ITEMS water_radius_x_m water_radius_z_m flat_radius_x_m flat_radius_z_m floor_y_m reference_scale_m)
    string(JSON ${field} GET "${habitat}" ${field})
endforeach()
configure_file("../pond-flexible-tail-check/habitat_config.h.in" habitat_config.h @ONLY)
add_executable(shore_fixtures shore-fixtures.cpp "${native}/core/checkpoint.cpp")
target_include_directories(shore_fixtures PRIVATE "${native}" "${native}/nn")
target_compile_features(shore_fixtures PRIVATE cxx_std_17)
foreach(target IN ITEMS installed_eval depth_eval)
    add_executable(${target} "${chooser}/rate-evaluate.cpp" "${native}/core/checkpoint.cpp")
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE "${native}" "${native}/nn" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(${target} PRIVATE CUDARL_RELIABLE_CHOOSER=1 CUDARL_APPENDED_COVERAGE=1)
    target_link_libraries(${target} PRIVATE Threads::Threads)
endforeach()
target_compile_definitions(depth_eval PRIVATE CUDARL_THREE_DEPTHS=1)
add_executable(depth_ledger depth-ledger.cpp "${native}/core/checkpoint.cpp")
target_compile_features(depth_ledger PRIVATE cxx_std_17)
target_include_directories(depth_ledger PRIVATE "${native}" "${native}/nn" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(depth_ledger PRIVATE CUDARL_RELIABLE_CHOOSER=1 CUDARL_APPENDED_COVERAGE=1 CUDARL_THREE_DEPTHS=1)
