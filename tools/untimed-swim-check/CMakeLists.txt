cmake_minimum_required(VERSION 3.24)
project(UntimedSwimBookkeepingCheck LANGUAGES CXX)
get_filename_component(worktree "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(untimed_swim_bookkeeping_check bookkeeping_check.cpp)
target_include_directories(untimed_swim_bookkeeping_check PRIVATE "${worktree}/src" "${worktree}/src/nn")
if(MSVC)
  target_compile_options(untimed_swim_bookkeeping_check PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(untimed_swim_bookkeeping_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
