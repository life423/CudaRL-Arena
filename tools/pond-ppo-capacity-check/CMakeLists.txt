cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120)
endif()
project(PondCategoricalCapacityCheck LANGUAGES CXX CUDA)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(source_root "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
set(old_math "${CMAKE_CURRENT_SOURCE_DIR}/../../build/flexible-tail/phase-aware-v4/source/src/rl/ppo_math.h")
configure_file("${old_math}" preserved_ppo_math.h COPYONLY)
add_executable(pond_ppo_capacity_check check.cu "${source_root}/rl/ppo_kernels.cu")
set_target_properties(pond_ppo_capacity_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES CUDA_RUNTIME_LIBRARY Static)
target_include_directories(pond_ppo_capacity_check PRIVATE "${source_root}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(pond_ppo_capacity_check PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
