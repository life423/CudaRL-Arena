cmake_minimum_required(VERSION 3.24)
project(PondResidualPpo LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
add_subdirectory(../pond-ppo "${CMAKE_BINARY_DIR}/ppo" EXCLUDE_FROM_ALL)
# The source manifest from the shared rollout includes the physical model,
# observation adapter, entropy/GAE/PPO, CUDA learning kernels and optimizer.
file(READ "${CMAKE_BINARY_DIR}/pond_ppo_identity.h" identity)
foreach(path IN ITEMS contract.h train.cu staged_schedule.h CMakeLists.txt identity.h.in)
    file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/${path}" digest)
    string(APPEND identity "${path}:${digest};")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/${path}")
endforeach()
string(SHA256 identity "${identity}")
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../../src/environment/pond_swept_clearance.h" sweep_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../../src/simulation/pond_runtime.h" runtime_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../../src/environment/pond_body_geometry.h" geometry_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../../src/environment/pond_geometry_v1.h" terrain_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../../src/cuda/pond_math_profile.h" math_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../../src/cuda/pond_geometry_inspect.cu" inspect_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../../src/cuda/pond_geometry_inspect_link.cu" inspect_link_hash)
configure_file(identity.h.in "${CMAKE_BINARY_DIR}/residual_identity.h" @ONLY)
add_executable(pond_residual_ppo train.cu ../pond-ppo/admission.cpp ../../src/core/checkpoint.cpp ../../src/nn/motor/saved_motor_contract.cpp)
target_include_directories(pond_residual_ppo PRIVATE "${CMAKE_BINARY_DIR}" "${CMAKE_BINARY_DIR}/ppo")
set_source_files_properties(train.cu PROPERTIES COMPILE_DEFINITIONS "NOMINMAX;WIN32_LEAN_AND_MEAN")
target_compile_options(pond_residual_ppo PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
pond_link_shared_inspection(pond_residual_ppo)
target_link_libraries(pond_residual_ppo PRIVATE pond_ppo_gpu_core bcrypt)
pond_warnings(pond_residual_ppo)
