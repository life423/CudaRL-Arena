cmake_minimum_required(VERSION 3.24)
project(PondGeometryCheck LANGUAGES CXX)
get_filename_component(worktree "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(pond_geometry_check geometry_check.cpp)
target_include_directories(pond_geometry_check PRIVATE "${worktree}/src")
if(MSVC)
  target_compile_options(pond_geometry_check PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(pond_geometry_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
