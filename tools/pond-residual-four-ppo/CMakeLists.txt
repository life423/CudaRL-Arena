cmake_minimum_required(VERSION 3.24)
project(PondResidualFourPpo LANGUAGES CXX CUDA)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
if(NOT TARGET pond_ppo_gpu_core)
    add_subdirectory(../pond-ppo "${CMAKE_BINARY_DIR}/ppo" EXCLUDE_FROM_ALL)
endif()
option(POND_BUILD_POSTURE_LESSON "Build the early upright turning lesson" OFF)
# The source manifest from the shared rollout includes the physical model,
# observation adapter, entropy/GAE/PPO, CUDA learning kernels and optimizer.
file(READ "${CMAKE_BINARY_DIR}/pond_ppo_identity.h" identity)
foreach(path IN ITEMS contract.h train.cu course_rate_reward.h turn_range_schedule.h
        ../pond-residual-ppo/staged_schedule.h CMakeLists.txt ../pond-residual-ppo/identity.h.in)
    file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/${path}" digest)
    string(APPEND identity "${path}:${digest};")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/${path}")
endforeach()
if(POND_BUILD_POSTURE_LESSON)
    string(APPEND identity "posture_lesson=ON;")
endif()
string(SHA256 identity "${identity}")
file(SHA256 "${repo}/src/environment/pond_swept_clearance.h" sweep_hash)
file(SHA256 "${repo}/src/simulation/pond_runtime.h" runtime_hash)
file(SHA256 "${repo}/src/environment/pond_body_geometry.h" geometry_hash)
file(SHA256 "${repo}/src/environment/pond_geometry_v1.h" terrain_hash)
file(SHA256 "${repo}/src/cuda/pond_math_profile.h" math_hash)
file(SHA256 "${repo}/src/cuda/pond_geometry_inspect.cu" inspect_hash)
file(SHA256 "${repo}/src/cuda/pond_geometry_inspect_link.cu" inspect_link_hash)
configure_file("../pond-residual-ppo/identity.h.in" residual_identity.h @ONLY)
set(native_support ../pond-ppo/admission.cpp ../../src/core/checkpoint.cpp ../../src/nn/motor/saved_motor_contract.cpp)
add_executable(pond_residual_four_ppo train.cu ${native_support})
set_target_properties(pond_residual_four_ppo PROPERTIES OUTPUT_NAME pond_residual_ppo)
target_include_directories(pond_residual_four_ppo PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_BINARY_DIR}")
set_source_files_properties(train.cu PROPERTIES COMPILE_DEFINITIONS "NOMINMAX;WIN32_LEAN_AND_MEAN")
target_compile_options(pond_residual_four_ppo PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
pond_link_shared_inspection(pond_residual_four_ppo)
target_link_libraries(pond_residual_four_ppo PRIVATE pond_ppo_gpu_core bcrypt)
pond_warnings(pond_residual_four_ppo)

# Both programs link the same compiled GPU math and geometry libraries.
file(READ "${CMAKE_CURRENT_BINARY_DIR}/residual_identity.h" replay_identity)
foreach(path IN ITEMS ../pond-residual-four-replay/replay.cu
        ../pond-collection-replay/driver.cpp ../pond-collection-replay/data.h
        ../pond-collection-replay/replay_identity.h.in)
    file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/${path}" digest)
    string(APPEND replay_identity "${path}:${digest};")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/${path}")
endforeach()
string(SHA256 replay_identity "${replay_identity}")
configure_file("../pond-collection-replay/replay_identity.h.in" collection_replay_identity.h @ONLY)
add_executable(pond_residual_four_replay ../pond-residual-four-replay/replay.cu ../pond-collection-replay/driver.cpp ${native_support})
target_include_directories(pond_residual_four_replay PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_BINARY_DIR}")
target_compile_definitions(pond_residual_four_replay PRIVATE POND_COLLECTION_FOUR_OUTPUT NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(pond_residual_four_replay PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
set_property(SOURCE ../pond-residual-four-replay/replay.cu APPEND PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_BINARY_DIR}/residual_identity.h;${CMAKE_CURRENT_LIST_DIR}/train.cu")
pond_link_shared_inspection(pond_residual_four_replay)
target_link_libraries(pond_residual_four_replay PRIVATE pond_ppo_gpu_core bcrypt)
if(POND_BUILD_POSTURE_LESSON)
    set_target_properties(pond_residual_four_ppo PROPERTIES OUTPUT_NAME pond_turn_ppo_posture_v1)
    set_target_properties(pond_residual_four_replay PROPERTIES OUTPUT_NAME pond_turn_posture_replay_v1)
    target_compile_definitions(pond_residual_four_ppo PRIVATE POND_POSTURE_LESSON)
    target_compile_definitions(pond_residual_four_replay PRIVATE POND_POSTURE_LESSON)
endif()
add_custom_target(qualified_four_bins DEPENDS pond_residual_four_ppo pond_residual_four_replay)
