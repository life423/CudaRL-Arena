cmake_minimum_required(VERSION 3.24)
project(PondBodyGeometryCheck LANGUAGES CXX)
option(CUDARL_BODY_GEOMETRY_CUDA "Build the isolated fixed-state CUDA geometry check" OFF)
get_filename_component(worktree "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(pond_body_geometry_check body_geometry_check.cpp)
target_include_directories(pond_body_geometry_check PRIVATE "${worktree}/src" "${worktree}/src/nn")
if(MSVC)
  target_compile_options(pond_body_geometry_check PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(pond_body_geometry_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(CUDARL_BODY_GEOMETRY_CUDA)
  set(CMAKE_CUDA_ARCHITECTURES 120)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(pond_body_geometry_cuda_check body_geometry_cuda_check.cu)
  target_include_directories(pond_body_geometry_cuda_check PRIVATE "${worktree}/src" "${worktree}/src/nn")
  target_compile_options(pond_body_geometry_cuda_check PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(pond_body_geometry_cuda_check PRIVATE CUDA::cudart_static)
endif()
